状态模型 · TransferItem

单一事实来源放进 useTransferStore；进度只在 transferring 阶段随确认字节增长。

type TransferStatus =
  | 'queued'              // 等待开始
  | 'waiting_for_target'  // 等待已连接设备
  | 'connecting'          // 正在建立连接 · 进度 0%
  | 'ready'               // 准备发送 · 进度 0%
  | 'transferring'        // 正在发送 · 进度增长
  | 'completed'           // 发送成功 · 收到对方 ack
  | 'failed'              // 发送失败 · 保留进度可重试
  | 'cancelled';          // 已取消

type TransferItem = {
  id: string;  fileName: string;  fileSize: number;
  targetDeviceId?: string;  sessionId?: string;
  status: TransferStatus;  progress: number;   // 0..1
  sentBytes: number;  acknowledgedBytes: number;
  transport: 'lan' | 'remote' | 'turn';
  errorMessage?: string;  createdAt: string;
};
